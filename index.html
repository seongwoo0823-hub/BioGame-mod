🟢 실시간 P2P 대기
순위 학번·이름 접속 웹·기기 환경 종합 점수 플레이 게임 수 선택 종목 기록 / 종목별 요약 최근 갱신
💡 같은 학급 코드를 입력한 학생들은 서로 다른 기기(PC·태블릿·스마트폰)에서도 P2P로 기록이 실시간 공유됩니다. ☁️ 클라우드 DB: 미연결 (P2P + 로컬 저장 활성)

☁️ 어떤 웹 환경에서 플레이하든 24시간 영구 기록되는 구글 스프레드시트 연동

선생님 PC가 꺼져 있거나 학생들이 각자 집·스마트폰에서 과제로 플레이할 때도 모든 학생의 점수와 접속 환경을 한곳에 자동 누적하려면 구글 스프레드시트(무료)를 연결하세요. 별도 유료 서버 없이 1분이면 설정됩니다!
📌 구글 시트 → [확장 프로그램] → [Apps Script]에 아래 코드를 붙여넣고 [배포 → 새 배포 → 웹 앱(모든 사용자)]로 배포하세요: 
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  var rows = sheet.getDataRange().getValues();
  var foundRow = -1;
  for (var i = 1; i < rows.length; i++) {
    if (String(rows[i][0]) === String(data.uid)) { foundRow = i + 1; break; }
  }
  var rowData = [data.uid, data.stId, data.name, data.env, data.totalScore, data.playedCount, JSON.stringify(data.games), new Date().toISOString()];
  if (foundRow > 0) sheet.getRange(foundRow, 1, 1, rowData.length).setValues([rowData]);
  else {
    if (rows.length === 0 || rows[0][0] !== 'UID') sheet.appendRow(['UID','학번','이름','웹환경','종합점수','플레이수','게임별상세JSON','갱신시각']);
    sheet.appendRow(rowData);
  }
  return ContentService.createTextOutput(JSON.stringify({ok:true})).setMimeType(ContentService.MimeType.JSON);
}
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var out = [];
  for (var i = 1; i < rows.length; i++) {
    if (!rows[i][0]) continue;
    var g = {}; try { g = JSON.parse(rows[i][6] || '{}'); } catch(_) {}
    out.push({uid:rows[i][0], stId:rows[i][1], name:rows[i][2], env:rows[i][3], totalScore:Number(rows[i][4])||0, playedCount:Number(rows[i][5])||0, games:g, updatedAt:rows[i][7]});
  }
  return ContentService.createTextOutput(JSON.stringify({ok:true, records:out})).setMimeType(ContentService.MimeType.JSON);
}
